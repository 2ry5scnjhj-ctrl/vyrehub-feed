-- Sauvegarde / chargement
local function saveState()
    if not writefile then return end
    pcall(function()
        writefile(FILE, HttpService:JSONEncode({
            found = State.found,
            visited = State.visited,
            running = Config.Running,
            config = {
                Targets = Config.Targets,
                MaxPlayers = Config.MaxPlayers,
                ScanDelay = Config.ScanDelay,
                StopOnFind = Config.StopOnFind,
                FeedURL = Config.FeedURL,
                Key = Config.Key,
                Mode = Config.Mode,
                AutoJoin = Config.AutoJoin,
            },
        }))
    end)
end

local resumeRun = false
local function loadState()
    if not (readfile and isfile) then return end
    local ok, data = pcall(function()
        if isfile(FILE) then
            return HttpService:JSONDecode(readfile(FILE))
        end
    end)
    if ok and type(data) == "table" then
        if type(data.found) == "table" then State.found = data.found end
        if type(data.visited) == "table" then State.visited = data.visited end
        State.visited[game.JobId] = true
        if type(data.config) == "table" then
            local c = data.config
            Config.Targets = c.Targets or Config.Targets
            Config.MaxPlayers = c.MaxPlayers or Config.MaxPlayers
            Config.ScanDelay = c.ScanDelay or Config.ScanDelay
            if c.StopOnFind ~= nil then Config.StopOnFind = c.StopOnFind end
            Config.FeedURL = c.FeedURL or Config.FeedURL
            Config.Key = c.Key or Config.Key
            if c.Mode == "viewer" or c.Mode == "scanner" then Config.Mode = c.Mode end
            if c.AutoJoin ~= nil then Config.AutoJoin = c.AutoJoin end
        end
        resumeRun = data.running == true
    end
end
loadState()

-- Requête HTTP compatible : retourne (corps, code HTTP)
local function httpRequest(method, url, body)
    local req = (syn and syn.request) or http_request or request
    if req then
        local ok, res = pcall(req, {
            Url = url,
            Method = method,
            Headers = { ["Content-Type"] = "application/json" },
            Body = body,
        })
        if ok and res then
            return res.Body, res.StatusCode
        end
    end
    if method == "GET" then
        local ok, b = pcall(function() return game:HttpGet(url) end)
        if ok and b then return b, 200 end
    end
    return nil, nil
end

local function httpGet(url)
    return httpRequest("GET", url)
end

-- Liste des mots-clés
local function getTargets()
    local list = {}
    for word in string.gmatch(Config.Targets, "[^,]+") do
        word = word:gsub("^%s+", ""):gsub("%s+$", "")
        if word ~= "" then table.insert(list, string.lower(word)) end
    end
    return list
end

-- Scan du serveur actuel : retourne une table {nom = quantité}
local function scanServer()
    local targets = getTargets()
    local found = {}
    local total = 0
    if #targets == 0 then return found, total end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("Model") or obj:IsA("BasePart") then
            local lname = string.lower(obj.Name)
            for _, t in ipairs(targets) do
                if string.find(lname, t, 1, true) then
                    local owner = Players:GetPlayerFromCharacter(obj)
                    local inChar = false
                    for _, p in ipairs(Players:GetPlayers()) do
                        if p.Character and obj:IsDescendantOf(p.Character) then
                            inChar = true
                            break
                        end
                    end
                    if not owner and not inChar then
                        found[obj.Name] = (found[obj.Name] or 0) + 1
                        total = total + 1
                    end
                    break
                end
            end
        end
    end
    return found, total
end

-- Recherche d'un serveur : retourne (jobId) ou (nil, raison)
local function findServer()
    local lastErr = nil
    local sawAnyServer = false
    local fallback = nil

    for _, host in ipairs(HOSTS) do
        local cursor = ""
        for _ = 1, 5 do
            local url = ("https://%s/v1/games/%d/servers/Public?sortOrder=Asc&limit=100%s")
                :format(host, game.PlaceId, cursor ~= "" and ("&cursor=" .. cursor) or "")
            local body, code = httpGet(url)

            if not body then
                lastErr = "HTTP bloqué (" .. host .. ")"
                break
            end
            if code == 429 then
                lastErr = "limite 429 (" .. host .. ")"
                break
            end
            if code and code ~= 200 then
                lastErr = "HTTP " .. tostring(code) .. " (" .. host .. ")"
                break
            end

            local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
            if not ok or type(data) ~= "table" or type(data.data) ~= "table" then
                lastErr = "réponse invalide (" .. host .. ")"
                break
            end

            for _, server in ipairs(data.data) do
                if server.id and server.id ~= game.JobId
                    and server.playing and server.maxPlayers
                    and server.playing < server.maxPlayers then
                    sawAnyServer = true
                    if not State.visited[server.id] then
                        if server.playing <= Config.MaxPlayers then
                            return server.id
                        end
                        if not fallback then fallback = server.id end
                    end
                end
            end

            if data.nextPageCursor then
                cursor = data.nextPageCursor
            else
                break
            end
        end

        if fallback then return fallback end
    end

    if fallback then return fallback end

    if sawAnyServer then
        State.visited = {}
        State.visited[game.JobId] = true
        saveState()
        return nil, "tous visités, réinitialisation"
    end

    return nil, lastErr or "aucun serveur disponible"
end

local function joinServer(jobId)
    if jobId == game.JobId then
        setStatus("tu es déjà dans ce serveur", COL.yellow)
        return
    end
    State.visited[jobId] = true
    saveState()
    setStatus("téléportation vers le serveur...", COL.accent)
    local ok, err = pcall(function()
        TeleportService:TeleportToPlaceInstance(game.PlaceId, jobId, LocalPlayer)
    end)
    if not ok then
        warn("[VyreHub] Erreur téléportation : " .. tostring(err))
        setStatus("échec de la téléportation", COL.red)
    end
end

-- Envoi d'une trouvaille au feed (mode scanner)
local function reportToFeed(entry)
    if Config.FeedURL == "" then return end
    task.spawn(function()
        local payload = HttpService:JSONEncode({
            key = Config.Key,
            jobId = entry.jobId,
            placeId = game.PlaceId,
            pets = entry.pets,
            players = entry.players,
        })
        local body, code = httpRequest("POST", Config.FeedURL .. "/report", payload)
        if not body then
            warn("[VyreHub] Envoi au feed impossible")
        elseif code and code ~= 200 then
            warn("[VyreHub] Feed a répondu HTTP " .. tostring(code) .. " (clé incorrecte ?)")
        end
    end)
end

-- Ajoute / remplace une entrée dans la liste
local function upsertEntry(entry)
    for i = #State.found, 1, -1 do
        if State.found[i].jobId == entry.jobId then
            table.remove(State.found, i)
        end
    end
    table.insert(State.found, entry)
    while #State.found > 50 do table.remove(State.found, 1) end
end

-- Trouvaille locale (mode scanner)
local function addFound(petsTable)
    local names = {}
    for name, qty in pairs(petsTable) do
        table.insert(names, qty > 1 and (name .. " x" .. qty) or name)
    end
    table.sort(names)

    local entry = {
        pets = names,
        jobId = game.JobId,
        players = #Players:GetPlayers() .. "/" .. Players.MaxPlayers,
        time = os.time(),
    }
    upsertEntry(entry)
    renderList()
    saveState()
    reportToFeed(entry)
end

            local delta = input.Position - dragStart
            main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X,
                startPos.Y.Scale, startPos.Y.Offset + delta.Y)
        end
    end)
    UserInputService.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseButton1
            or input.UserInputType == Enum.UserInputType.Touch then
            dragging = false
        end
    end)
end

-- ===================== Boucles principales =====================
local function setRunningUI(running)
    startBtn.Text = running and "■ Arrêter" or "▶ Démarrer"
    startBtn.BackgroundColor3 = running and COL.red or COL.accent
end

-- MODE SCANNER : saute de serveur en serveur, scanne, envoie au feed
local function scannerLoop()
    task.wait(Config.ScanDelay)
    while Config.Running and gui.Parent do
        setStatus("scan du serveur...", COL.yellow)
        local found, total = scanServer()
        if total > 0 then
            addFound(found)
            if Config.StopOnFind then
                Config.Running = false
                setRunningUI(false)
                saveState()
                setStatus("TROUVÉ ! (" .. total .. " objet(s))", COL.green)
                return
            end
        end

        setStatus("recherche d'un serveur...", COL.yellow)
        local serverId, reason = findServer()
        if serverId then
            State.visited[serverId] = true
            saveState()
            setStatus("téléportation...", COL.accent)
            local ok, err = pcall(function()
                TeleportService:TeleportToPlaceInstance(game.PlaceId, serverId, LocalPlayer)
            end)
            if not ok then
                warn("[VyreHub] Erreur téléportation : " .. tostring(err))
                setStatus("échec TP : " .. tostring(err), COL.red)
            end
            task.wait(5)
        else
            warn("[VyreHub] Pas de serveur : " .. tostring(reason))
            setStatus(tostring(reason) .. " - nouvel essai...", COL.red)
            task.wait(5)
        end
    end
end

-- MODE VIEWER : interroge le feed et affiche les nouveaux pets
local lastSeen = 0

local function matchesTargets(entry)
    local targets = getTargets()
    if #targets == 0 then return true end
    for _, petName in ipairs(entry.pets or {}) do
        local lname = string.lower(tostring(petName))
        for _, t in ipairs(targets) do
            if string.find(lname, t, 1, true) then return true end
        end
    end
    return false
end

-- Retourne (nouvelles entrées, erreur)
local function pollFeed()
    local url = Config.FeedURL .. "/feed?since=" .. tostring(lastSeen)
    local body, code = httpRequest("GET", url)
    if not body then return nil, "feed injoignable (HTTP bloqué ou URL fausse)" end
    if code and code ~= 200 then return nil, "feed HTTP " .. tostring(code) end

    local ok, data = pcall(function() return HttpService:JSONDecode(body) end)
    if not ok or type(data) ~= "table" then return nil, "réponse du feed invalide" end

    local fresh = {}
    for _, e in ipairs(data) do
        if type(e) == "table" and e.jobId and type(e.pets) == "table" then
            if type(e.time) == "number" and e.time > lastSeen then lastSeen = e.time end
            if e.placeId == game.PlaceId and matchesTargets(e) then
                local entry = {
                    pets = e.pets,
                    jobId = e.jobId,
                    players = e.players,
                    time = math.floor((e.time or 0) / 1000),
                }
                upsertEntry(entry)
                if e.jobId ~= game.JobId and not State.visited[e.jobId] then
                    table.insert(fresh, entry)
                end
            end
        end
    end
    return fresh
end

local function viewerLoop()
    if Config.FeedURL == "" then
        Config.Running = false
        setRunningUI(false)
        setStatus("mets l'URL du feed dans Réglages", COL.red)
        saveState()
        return
    end
    while Config.Running and gui.Parent do
        local fresh, err = pollFeed()
        if err then
            setStatus(err, COL.red)
        else
            if #fresh > 0 then
                renderList()
                saveState()
                setStatus(#fresh .. " nouveau(x) pet(s) !", COL.green)
                if Config.AutoJoin then
                    local newest = fresh[#fresh]
                    Config.Running = false
                    setRunningUI(false)
                    saveState()
                    joinServer(newest.jobId)
                    return
                end
            else
                setStatus("écoute du feed...", COL.yellow)
            end
        end
        task.wait(2)
    end
end

local function startLoop()
    if Config.Mode == "viewer" then
        task.spawn(viewerLoop)
    else
        task.spawn(scannerLoop)
    end
end
